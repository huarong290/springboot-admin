package com.springboot.admin.service.impl;

import com.springboot.admin.service.IRedisService;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.data.redis.connection.RedisConnection;
import org.springframework.data.redis.core.*;
import org.springframework.data.redis.core.script.DefaultRedisScript;
import org.springframework.stereotype.Service;

import java.nio.charset.StandardCharsets;
import java.util.*;
import java.util.concurrent.TimeUnit;

/**
 * Redis 全能服务实现类 (Enterprise Edition)
 *
 * <p>
 * <h3>核心设计理念：</h3>
 * <ul>
 * <li><b>Fail-Secure (故障阻断):</b> 统一封装 try-catch，Redis 宕机时自动降级返回默认值 (null/false)，保护主业务不受影响。</li>
 * <li><b>Atomic (原子性):</b> 关键业务（如限流、一次性 Token）内置 Lua 脚本，保证操作原子性。</li>
 * <li><b>Non-Blocking (非阻塞):</b> 使用 SCAN 替代 KEYS 命令，安全处理海量数据查询。</li>
 * <li><b>DRY (拒绝重复):</b> 采用函数式编程 (executeSafe) 消除冗余的异常处理代码。</li>
 * </ul>
 * </p>
 *
 * @author YourTeam
 */
@Slf4j
@Service
@RequiredArgsConstructor
public class RedisServiceImpl implements IRedisService {

    // 1. 注入 StringRedisTemplate (用于常规 K-V 操作，性能最高)
    private final StringRedisTemplate stringRedisTemplate;

    // 2. 注入 RedisTemplate (用于复杂对象存储，序列化规则见 RedisConfig)
    private final RedisTemplate<String, Object> redisTemplate;

    // ==========================================================
    // Lua 脚本集中管理 (Static Final, 避免重复编译)
    // ==========================================================

    /** 脚本：原子自增并设置过期时间 (用于限流、防刷) */
    private static final DefaultRedisScript<Long> INCR_WITH_EXPIRE_SCRIPT;

    /** 脚本：原子获取并删除 (用于一次性 Token、验证码) */
    private static final DefaultRedisScript<String> GET_AND_DELETE_SCRIPT;

    static {
        // 初始化 Lua 脚本
        INCR_WITH_EXPIRE_SCRIPT = new DefaultRedisScript<>(
                "local current = redis.call('INCR', KEYS[1]) " +
                "if current == 1 then redis.call('PEXPIRE', KEYS[1], ARGV[1]) end " +
                "return current", Long.class
        );

        GET_AND_DELETE_SCRIPT = new DefaultRedisScript<>(
                "local v = redis.call('GET', KEYS[1]) " +
                "if not v then return nil end " +
                "redis.call('DEL', KEYS[1]) " +
                "return v", String.class
        );
    }

    // ==========================================================
    // 1. 通用 Key 操作
    // ==========================================================

    @Override
    public boolean expire(String key, long timeout, TimeUnit unit) {
        return executeSafe(() -> Boolean.TRUE.equals(stringRedisTemplate.expire(key, timeout, unit)), key, false);
    }

    @Override
    public long getExpire(String key, TimeUnit unit) {
        return executeSafe(() -> {
            Long expire = stringRedisTemplate.getExpire(key, unit);
            return expire != null ? expire : -2;
        }, key, -2L);
    }

    @Override
    public boolean hasKey(String key) {
        return executeSafe(() -> Boolean.TRUE.equals(stringRedisTemplate.hasKey(key)), key, false);
    }

    @Override
    public boolean deleteKey(String key) {
        return executeSafe(() -> Boolean.TRUE.equals(stringRedisTemplate.delete(key)), key, false);
    }

    @Override
    public boolean deleteKeys(Collection<String> keys) {
        if (keys == null || keys.isEmpty()) return false;
        return executeSafe(() -> {
            Long count = stringRedisTemplate.delete(keys);
            return count != null && count > 0;
        }, "batch-delete-size:" + keys.size(), false);
    }

    // ==========================================================
    // 2. String (字符串) 操作
    // ==========================================================

    @Override
    public boolean setValue(String key, String value) {
        return executeSafe(() -> {
            stringRedisTemplate.opsForValue().set(key, value);
            return true;
        }, key, false);
    }

    @Override
    public boolean setValue(String key, String value, long timeout, TimeUnit unit) {
        return executeSafe(() -> {
            stringRedisTemplate.opsForValue().set(key, value, timeout, unit);
            return true;
        }, key, false);
    }

    @Override
    public String getValue(String key) {
        return executeSafe(() -> stringRedisTemplate.opsForValue().get(key), key, null);
    }

    @Override
    public List<String> multiGet(Collection<String> keys) {
        if (keys == null || keys.isEmpty()) return Collections.emptyList();
        return executeSafe(
            () -> stringRedisTemplate.opsForValue().multiGet(keys),
            "multiGet-size:" + keys.size(),
            Collections.emptyList()
        );
    }

    @Override
    public Long increment(String key) {
        return executeSafe(() -> stringRedisTemplate.opsForValue().increment(key), key, null);
    }

    /**
     * 原子递增并设置过期
     * <p>注意：unit.toMillis(timeout) 确保传给 Lua 的是毫秒</p>
     */
    @Override
    public Long increment(String key, long timeout, TimeUnit unit) {
        return executeSafe(() -> stringRedisTemplate.execute(
                INCR_WITH_EXPIRE_SCRIPT,
                Collections.singletonList(key),
                String.valueOf(unit.toMillis(timeout))
        ), key, null);
    }

    @Override
    public String getAndDelete(String key) {
        return executeSafe(() -> stringRedisTemplate.execute(
                GET_AND_DELETE_SCRIPT,
                Collections.singletonList(key)
        ), key, null);
    }

    // ==========================================================
    // 2.1 Object (对象) 操作 - 使用 JSON 序列化
    // ==========================================================

    @Override
    public boolean setObject(String key, Object value, long timeout, TimeUnit unit) {
        return executeSafe(() -> {
            redisTemplate.opsForValue().set(key, value, timeout, unit);
            return true;
        }, key, false);
    }

    @Override
    public <T> T getObject(String key, Class<T> clazz) {
        return executeSafe(() -> {
            Object obj = redisTemplate.opsForValue().get(key);
            return obj != null ? clazz.cast(obj) : null;
        }, key, null);
    }

    // ==========================================================
    // 3. Hash (哈希表) 操作
    // ==========================================================

    @Override
    public boolean hSet(String key, String hashKey, String value) {

    @Override
    public boolean hSet(String key, String hashKey, String value, long timeout, TimeUnit unit) {
        return executeSafe(() -> {
            stringRedisTemplate.opsForHash().put(key, hashKey, value);
            return expire(key, timeout, unit);
        }, key, false);
    }

    @Override
    public boolean hMSet(String key, Map<String, String> map) {

    @Override
    public boolean hMSet(String key, Map<String, String> map, long timeout, TimeUnit unit) {

        }, key, false);
    }

    @Override
    public String hGet(String key, String hashKey) {
        return executeSafe(() -> {

            return val != null ? val.toString() : null;

    }

    @Override
    public Map<Object, Object> hGetAll(String key) {
        return executeSafe(() -> stringRedisTemplate.opsForHash().entries(key), key, Collections.emptyMap());

    @Override
    public boolean hDel(String key, Object... hashKeys) {
        return executeSafe(() -> {

    }

    @Override
    public boolean hHasKey(String key, String hashKey) {
        return executeSafe(() -> stringRedisTemplate.opsForHash().hasKey(key, hashKey), key, false);
    }

    @Override
    public Long hIncr(String key, String hashKey, long delta) {

    }

    // ==========================================================
    // 4. List (列表) 操作

    @Override

    // ==========================================================
    // 5. Set (集合) 操作
    // ==========================================================

    @Override
    public boolean sAdd(String key, String... values) {

    @Override
    public boolean sAddWithExpire(String key, long timeout, TimeUnit unit, String... values) {
        return executeSafe(() -> {
            Long count = stringRedisTemplate.opsForSet().add(key, values);

            return count != null && count > 0;
        }, key, false);
    }

    // ==========================================================
    // 7. 高级操作 (Scan 非阻塞扫描)
    // ==========================================================

    @Override
    public Set<String> scan(String pattern) {
        return executeSafe(() -> {
            // 使用 execute 显式获取 RedisConnection，避免 Lambda 类型推断歧义
            return stringRedisTemplate.execute((RedisConnection connection) -> {
                Set<String> keys = new HashSet<>();
                ScanOptions options = ScanOptions.scanOptions().match(pattern).count(1000).build();
                try (Cursor<byte[]> cursor = connection.scan(options)) {
                    while (cursor.hasNext()) {
                        // 显式指定 UTF-8 编码，防止乱码
                        keys.add(new String(cursor.next(), StandardCharsets.UTF_8));
                    }
                }
                return keys;
            });
        }, pattern, Collections.emptySet());
    }

    // ==========================================================
    // 8. 私有辅助方法 (Fail-Secure 统一异常熔断)
    // ==========================================================

    /**
     * 安全执行器 (返回类型: T)
     * @param supplier 业务逻辑
     * @param key 相关的 Redis Key (用于日志记录)
     * @param defaultValue 发生异常时的默认返回值
     */
    private <T> T executeSafe(SupplierWithException<T> supplier, String key, T defaultValue) {
        try {
            return supplier.get();
        } catch (Exception e) {
            handleException(key, e);
            return defaultValue;
        }
    }

    /**
     * 智能日志降噪处理
     * <p>
     * 策略：
     * 1. Debug 模式：打印完整堆栈，方便开发排查。
     * 2. 生产环境 (Info/Error)：只打印 Key 和 错误信息，避免 Redis 宕机时堆栈刷屏导致磁盘 IO 飙升。
     * </p>
     */
    private void handleException(String key, Exception e) {
        if (log.isDebugEnabled()) {
            log.error("Redis Op Failed. Key: {}", key, e);
        } else {
            log.error("Redis Op Failed. Key: {}. Error: {}", key, e.getMessage());
        }
    }

    /**
     * 支持抛出异常的函数式接口
     */
    @FunctionalInterface
    private interface SupplierWithException<T> {
        T get() throws Exception;
    }
}
